# ListPostsMetadata

Cover image, tags, excerpt, authors and other site metadata. Set for blog posts and changelogs, null for every other content type.

## Example Usage

```typescript
import { ListPostsMetadata } from "@usenotra/sdk/models/operations";

let value: ListPostsMetadata = {
  coverImage: {
    url: "https://cdn.example.com/covers/ship-notes.png",
    alt: "Screenshot of the new editor",
  },
  tags: [
    "editor",
    "performance",
  ],
  excerpt: "The editor now opens twice as fast on long posts.",
  authors: [
    {
      id: "auth_8f2kq1",
      slug: "jane-doe",
      name: "Jane Doe",
      title: "Head of Product",
      avatarUrl: "https://cdn.example.com/avatars/jane.png",
      url: "https://janedoe.dev",
      x: "https://x.com/janedoe",
      linkedin: "<value>",
      github: "<value>",
      youtube: "<value>",
      bluesky: "<value>",
      mastodon: "<value>",
      threads: "<value>",
      instagram: "<value>",
    },
  ],
  featured: false,
  noindex: false,
  hidden: false,
  displayDate: "2026-10-01",
};
```
